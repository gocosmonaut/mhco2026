# Hostinger public_html/.htaccess for Drupal 10 Composer Subdirectory (/web)
# ==============================================================================

<IfModule mod_rewrite.c>
  RewriteEngine On

  # ----------------------------------------------------------------------------
  # 1. Force HTTPS
  # ----------------------------------------------------------------------------
  RewriteCond %{HTTPS} off
  RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

  # ----------------------------------------------------------------------------
  # 2. Route Traffic to the /web Subdirectory
  # ----------------------------------------------------------------------------

  # CURRENT DOMAIN: darkorchid-seahorse-691577.hostingersite.com
  # Matches the domain with or without 'www.'
  RewriteCond %{HTTP_HOST} ^(www\.)?darkorchid-seahorse-691577\.hostingersite\.com$ [NC]
  RewriteCond %{REQUEST_URI} !^/web/
  RewriteRule ^(.*)$ /web/$1 [L]
